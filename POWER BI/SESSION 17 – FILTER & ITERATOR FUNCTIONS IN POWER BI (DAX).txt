


SESSION 17 – FILTER & ITERATOR FUNCTIONS IN POWER BI (DAX)

TASK 1 – FILTER()
Products Rating Above 4 =
COUNTROWS(
    FILTER(
        Flipkart,
        Flipkart[Rating] > 4
    )
)

TASK 2 – CALCULATE() + ALL()
Overall Average Order Value =
CALCULATE(
    AVERAGE(Zomato[OrderValue]),
    ALL(
        Zomato[City],
        Zomato[Cuisine]
    )
)

TASK 3 – RANKX()
Total Box Office =
SUM(Movies[BoxOffice])

Movie Rank =
RANKX(
    ALL(Movies[Movie]),
    [Total Box Office],
    ,
    DESC,
    DENSE
)

TASK 4 – SUMX()
Cart Total =
SUMX(
    Myntra,
    Myntra[Quantity] * Myntra[Price]
)

TASK 5 – AVERAGEX()
Average Restaurant Rating =
AVERAGEX(
    VALUES(RestaurantReviews[Restaurant]),
    CALCULATE(
        AVERAGE(RestaurantReviews[Rating])
    )
)

FUNCTION SUMMARY
FILTER() = Filter rows
CALCULATE() = Change filter context
ALL() = Remove filters
ALLEXCEPT() = Remove all filters except specified columns
VALUES() = Return unique values
RANKX() = Rank values
SUMX() = Calculate row by row and then sum
AVERAGEX() = Calculate row by row and then average
